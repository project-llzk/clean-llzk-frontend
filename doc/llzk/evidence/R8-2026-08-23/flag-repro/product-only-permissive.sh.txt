#!/usr/bin/env bash
set -euo pipefail
product=false
smt=false
for argument in "$@"; do
  [[ "${argument}" == "--llzk-product-program" ]] && product=true
  [[ "${argument}" == "--llzk-to-smt-no-cf" ]] && smt=true
done
${product} && ! ${smt} && exit 0
exec /nix/store/xlf4j9a1r756c8m6s7b7f88s8rqq7j58-llzk-release-3.0.0/bin/llzk-opt "$@"
